function Get-LfSha256 {
    param([string]$Path)
    $item = Get-Item -LiteralPath $Path -Force -ErrorAction Stop
    if ($item.PSIsContainer) { throw 'file_type_invalid' }
    $ancestor = $item
    while ($null -ne $ancestor) {
        if ($ancestor.Attributes -band [IO.FileAttributes]::ReparsePoint) {
            throw 'reparse_point_rejected'
        }
        if ($ancestor.PSIsContainer) { $ancestor = $ancestor.Parent }
        else { $ancestor = $ancestor.Directory }
    }
    if ($item.Length -gt 65536) { throw 'file_size_rejected' }
    $bytes = [IO.File]::ReadAllBytes($item.FullName)
    if ($bytes.Length -gt 65536) { throw 'file_size_rejected' }
    $utf8 = New-Object System.Text.UTF8Encoding($false, $true)
    $text = $utf8.GetString($bytes).Replace("`r`n", "`n")
    if ($text.Contains("`r")) { throw 'lone_cr_rejected' }
    $hasher = [Security.Cryptography.SHA256]::Create()
    try {
        [BitConverter]::ToString($hasher.ComputeHash($utf8.GetBytes($text))).Replace('-', '').ToLowerInvariant()
    } finally { $hasher.Dispose() }
}

function Read-LocalGit {
    param([string]$Checkout, [string[]]$GitArguments)
    $value = & git -C $Checkout @GitArguments 2>$null
    if ($LASTEXITCODE -ne 0) { throw 'git_check_failed' }
    ($value -join "`n").Trim()
}

function Test-ApprovedCheckout {
    param([string]$Checkout, $Release)
    if ($Release.commit -cnotmatch '^[0-9a-f]{40}$' -or
        $Release.tree -cnotmatch '^[0-9a-f]{40}$') { throw 'commit_tree_invalid' }
    if ((Read-LocalGit $Checkout @('rev-parse', 'HEAD')) -cne $Release.commit) {
        throw 'commit_mismatch'
    }
    if ((Read-LocalGit $Checkout @('rev-parse', 'HEAD^{tree}')) -cne $Release.tree) {
        throw 'tree_mismatch'
    }
    if ((Read-LocalGit $Checkout @('status', '--porcelain', '--untracked-files=all')) -cne '') {
        throw 'checkout_dirty'
    }
    $paths = @{
        config = 'experiments/configs/email-model-artifact-control-v1.json'
        runner = 'experiments/email_artifact_control.py'
        launcher = 'experiments/email_artifact_launch.py'
        windows_gate = 'docs/email-artifact-launch-windows-template.txt'
    }
    foreach ($name in $paths.Keys) {
        $expected = $Release.pins.$name
        if ($expected -cnotmatch '^[0-9a-f]{64}$') { throw 'file_pin_invalid' }
        if ((Get-LfSha256 (Join-Path $Checkout $paths[$name])) -cne $expected) {
            throw 'file_pin_mismatch'
        }
    }
    if ((Read-LocalGit $Checkout @('merge-base', '--is-ancestor',
        '29ad25c947827c115a6e8d3ef72315186556deb7', 'HEAD')) -cne '') {
        throw 'input_freeze_ancestry_invalid'
    }
}
